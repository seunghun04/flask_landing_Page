9. app.py 만들기

from flask import Flask, render_template

app = Flask(__name__)

@app.route("/main")
def main():
	return render_template("main.html")

@app.route("/about")
def about():
	return render_template("about.html")

if __name__ == "__main__" :
	app.run(debug=True)
